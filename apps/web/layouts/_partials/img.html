{{- /* Imagen del bundle de la página (page) o de assets/img/: AVIF y WebP a varios anchos (sin ampliar) con width/height para evitar saltos de maquetación. Argumentos: src, alt, page (opcional), sizes (opcional), eager (opcional, para la imagen que se ve al cargar). */ -}} {{- $sizes := .sizes | default "(min-width: 900px) 560px, 92vw" }} {{- with or (and .page (.page.Resources.Get .src)) (resources.Get (printf "img/%s" .src)) }} {{- $img := . }} {{- $max := cond (gt $img.Width 1200) 1200 $img.Width }} {{- $specs := slice }} {{- range slice 480 800 }}{{ if lt . $max }}{{ $specs = $specs | append (printf "%dx webp" .) }}{{ end }}{{ end }} {{- $specs = $specs | append (printf "%dx webp" $max) }} {{- $set := slice }} {{- $avifSet := slice }} {{- $last := "" }} {{- range $specs }} {{- $r := $img.Resize . }} {{- $set = $set | append (printf "%s %dw" $r.RelPermalink $r.Width) }} {{- $avif := $img.Resize (replace . "webp" "avif") }} {{- $avifSet = $avifSet | append (printf "%s %dw" $avif.RelPermalink $avif.Width) }} {{- $last = $r }} {{- end }} {{- else }} {{- errorf "img.html: %s not found in the page bundle or assets/img/" .src }} {{- end }}
